 2841: /**
 2842:  * Download a single remote file with optional retries. Dedicated servers often write savegame XML in a
 2843:  * burst; FTP during that window can see locked files, partial writes, or zero-byte reads — different
 2844:  * files fail on each poll. Retries with short backoff usually clear it.
 2845:  */
 2846: async function safeDownload(client, remotePath, localTmp, localFinal, options = {}) {
 2847:     const maxAttempts = Math.max(1, parseInt(options.maxAttempts, 10) || 1);
 2848:     const retryDelayMs = Math.max(0, parseInt(options.retryDelayMs, 10) || 300);
 2849:     let lastErr = '';
 2850:     for (let attempt = 1; attempt <= maxAttempts; attempt++) {
 2851:         try {
 2852:             try {
 2853:                 await fs.promises.unlink(localTmp);
 2854:             } catch (e) {
 2855:                 if (e && e.code !== 'ENOENT') {
 2856:                     /* ignore transient unlink issues like old behavior */
 2857:                 }
 2858:             }
 2859:             await client.downloadTo(localTmp, remotePath);
 2860:             let size = 0;
 2861:             try {
 2862:                 size = (await fs.promises.stat(localTmp)).size;
 2863:             } catch (e) {
 2864:                 lastErr = e && e.message ? String(e.message) : String(e);
 2865:                 if (attempt < maxAttempts) {
 2866:                     await new Promise((r) => setTimeout(r, retryDelayMs * attempt));
 2867:                 }
 2868:                 continue;
 2869:             }
 2870:             if (size > 0) {
 2871:                 try {
 2872:                     await fs.promises.unlink(localFinal);
 2873:                 } catch (e) {
 2874:                     if (e && e.code !== 'ENOENT') {
 2875:                         /* ignore */
 2876:                     }
 2877:                 }
 2878:                 await fs.promises.rename(localTmp, localFinal);
 2879:                 return true;
 2880:             }
 2881:             lastErr = 'empty or zero-byte after download';
 2882:         } catch (e) {
 2883:             lastErr = e && e.message ? String(e.message) : String(e);
 2884:         }
 2885:         if (attempt < maxAttempts) {
 2886:             await new Promise((r) => setTimeout(r, retryDelayMs * attempt));
 2887:         }
 2888:     }
 2889:     if (maxAttempts > 1 && lastErr && options.logFailures) {
 2890:         console.warn(`[FTP] ${path.basename(localFinal)}: failed after ${maxAttempts} tries — ${lastErr.slice(0, 120)}`);
 2891:     }
